create table public.panoramas (
  id uuid not null default gen_random_uuid (),
  city_id uuid not null,
  civic_object_id uuid not null,
  civic_object_type text not null,
  equirectangular_url text not null,
  thumbnail_url text null,
  label text null,
  description text null,
  display_order integer null default 0,
  status text not null default 'draft'::text,
  is_published boolean GENERATED ALWAYS as ((status = 'published'::text)) STORED null,
  created_by uuid null,
  updated_by uuid null,
  created_at timestamp with time zone null default now(),
  updated_at timestamp with time zone null default now(),
  constraint panoramas_pkey primary key (id)
) TABLESPACE pg_default;
